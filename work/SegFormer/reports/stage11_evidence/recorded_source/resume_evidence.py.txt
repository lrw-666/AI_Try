1. state check : rebuild from B/checkpoints/last.pt in this fresh process and compare
                 every restored object to the checkpoint payload, bitwise.
2. control     : A (uninterrupted) vs B (1 epoch, then resumed) - end-to-end.
3. isolation   : D vs E, two independent resumes from the *same* epoch-1 checkpoint,
                 which separates "resume" from "two different processes".

Tolerances are declared in DECLARED before any number is compared.
"""

DECLARED = {
    "state_at_load": "bitwise equality of model/optimizer/scaler/scheduler/LR/counters/RNG/generator",
    "batch_order_and_flip": "exact sequence equality",
    "lr_and_loss_after_boundary": "lr exact; loss differences reported, not gated",
    "model_after_control": "bitwise preferred; otherwise report max|diff| and the first divergent step",
    "preview_probabilities": "max|diff| <= 1e-5 and XOR == 0",
}


def ckpt(path):
    return torch.load(path, map_location="cpu", weights_only=False)


def tensor_diff(left, right):
    worst, key_worst = 0.0, None
    for key, value in left.items():
        other = right[key]
        diff = float((value.detach().float().cpu() - other.detach().float().cpu()).abs().max())
        if diff > worst:
            worst, key_worst = diff, key
    return {"max_abs_diff": worst, "worst_tensor": key_worst, "tensors": len(left)}


def optimizer_equal(left, right):
    if set(left["state"]) != set(right["state"]):
        return False, "state keys differ"
    for key, state in left["state"].items():
        other = right["state"][key]
        if set(state) != set(other):
            return False, f"fields differ at {key}"
        for field, value in state.items():
            if torch.is_tensor(value):
                if not torch.equal(value.detach().cpu(), other[field].detach().cpu()):
                    return False, f"{key}.{field}"
            elif value != other[field]:
                return False, f"{key}.{field}"
    return True, "identical"


def stream(path, min_epoch=None):
    rows = []
    for line in path.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        record = json.loads(line)
        if record.get("event") == "resume":
            continue
        if min_epoch is not None and record["epoch"] < min_epoch:
            continue
        rows.append(record)
    return rows


def trajectory(left_dir, right_dir, min_epoch):
    left = stream(left_dir / "updates.jsonl", min_epoch)
    right = stream(right_dir / "updates.jsonl", min_epoch)
    order = len(left) == len(right) and all(a["batch_ids"] == b["batch_ids"] for a, b in zip(left, right))
    flip = len(left) == len(right) and all(a["flipped_ids"] == b["flipped_ids"] for a, b in zip(left, right))
    lr_same = len(left) == len(right) and all(a["lr"] == b["lr"] for a, b in zip(left, right))
    diffs = [abs(a["total"] - b["total"]) for a, b in zip(left, right)]
    return {"rows": len(left), "batch_order_identical": order, "flip_identical": flip,
            "lr_identical": lr_same, "max_loss_abs_diff": max(diffs) if diffs else None,
            "loss_identical": bool(diffs) and max(diffs) == 0.0}


def probs(a_dir, b_dir):
    a, b = np.load(a_dir / "pre_save_probabilities.npz"), np.load(b_dir / "pre_save_probabilities.npz")
    keys = sorted(a.files)
    max_diff = max(float(np.abs(a[k] - b[k]).max()) for k in keys)
    xor = int(sum(int(((a[k] >= 0.5) != (b[k] >= 0.5)).sum()) for k in keys))
    return {"images": keys, "max_abs_diff": max_diff, "xor_pixels": xor,
            "total_pixels": int(sum(int(a[k].size) for k in keys)),
            "within_tolerance": max_diff <= 1e-5 and xor == 0}


# ---------------------------------------------------------------- 1. state check
config = load_config("configs/resume_fixture.yaml")
payload = read_checkpoint(RUNS / "resume_fixture_B" / "checkpoints" / "last.pt")
identities = freeze_identity(config, label="state-check",
                             extra_sources=("segformer_lab/train_full.py",))
restored = restore_checkpoint(payload, config, "cuda", identities=identities)
model_cmp = tensor_diff(payload["model"], restored["model"].state_dict())
opt_ok, opt_note = optimizer_equal(payload["optimizer"], restored["optimizer"].state_dict())
sched = restored["schedule"].state_dict()
state_check = {
    "checkpoint": str((RUNS / "resume_fixture_B" / "checkpoints" / "last.pt")),
    "model_bitwise_equal": model_cmp["max_abs_diff"] == 0.0,
    "model": model_cmp,
    "optimizer_equal": opt_ok, "optimizer_note": opt_note,
    "scaler_equal": payload["grad_scaler"] == restored["scaler"].state_dict(),
    "scheduler_equal": payload["scheduler"] == sched,
    "lr_equal": payload["lr"] == restored["lr"],
    "step_equal": payload["step"] == {"successful": restored["successful"],
                                      "attempts": restored["attempts"],
                                      "skipped": restored["skipped"]},
    "epoch_and_best_equal": (payload["next_epoch"] == restored["epoch_start"]
                             and payload["best"] == restored["best"]),
    "rng_restored_equals_payload": (
        torch.equal(torch.get_rng_state(), payload["rng"]["torch"])
        and all(torch.equal(a, b) for a, b in zip(torch.cuda.get_rng_state_all(), payload["rng"]["cuda"]))),
    "all_exact": bool(model_cmp["max_abs_diff"] == 0.0 and opt_ok
                      and payload["grad_scaler"] == restored["scaler"].state_dict()
                      and payload["scheduler"] == sched and payload["lr"] == restored["lr"]),
}
